#include <fstream>
#include <iostream>
#include <string>
using namespace std;

int main()
{
    ofstream outputFile;
    outputFile.open("Try.txt", ios::out);
    outputFile << "What do you say" << endl;
    outputFile << "I love you ";
    outputFile.close();

    ifstream Myfile("Try.txt");
    string text;
    while (getline(Myfile,text)){
        cout << text << endl;
    }
    Myfile.close();
    

    return 0;
}
